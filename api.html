# create next month's meetup from CI
curl -X POST https://api.localhost.events/v1/events \
  -H "Authorization: Bearer $LOCALHOST_TOKEN" \
  -d '{
    "title": "Demo Night #13",
    "city": "berlin",
    "starts_at": "2026-11-06T18:30:00+01:00",
    "venue": "Factory Berlin",
    "currency": "EUR",
    "tiers": [{ "name": "General", "price": 0, "capacity": 200 }],
    "cfp": { "open": true, "closes_at": "2026-10-30" }
  }'
